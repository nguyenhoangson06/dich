""" Nhận dạng giọng nói -> dịch sang tiếng Việt -> ghép phụ đề vào video. Cài đặt: pip install faster-whisper deep-translator # Cần cài FFmpeg (https://ffmpeg.org/download.html) và thêm vào PATH # Phần dịch dùng Google Translate nên cần có Internet. Cách dùng: python translate_subtitle.py video.mp4 python translate_subtitle.py video.mp4 --model medium --bilingual python translate_subtitle.py video.mp4 --src en --soft python translate_subtitle.py video.mp4 --color "#ffeb3b" --size 32 """ import argparse import subprocess from pathlib import Path from deep_translator import GoogleTranslator from faster_whisper import WhisperModel def format_time(seconds: float) -> str: ms = int(round(seconds * 1000)) h, ms = divmod(ms, 3_600_000) m, ms = divmod(ms, 60_000) s, ms = divmod(ms, 1000) return f"{h:02d}:{m:02d}:{s:02d},{ms:03d}" def transcribe(video: Path, model_size: str, device: str, src_lang: str | None): compute_type = "float16" if device == "cuda" else "int8" print(f"[1/3] Nhận dạng giọng nói (model '{model_size}', {device})...") model = WhisperModel(model_size, device=device, compute_type=compute_type) segments, info = model.transcribe( str(video), language=src_lang, # None = tự phát hiện ngôn ngữ vad_filter=True, beam_size=5, ) print(f" Ngôn ngữ phát hiện: {info.language} (độ tin cậy {info.language_probability:.0%})") # segments là generator -> chuyển thành list để dùng lại return [(s.start, s.end, s.text.strip()) for s in segments], info.language def translate_segments(segments, src_lang: str): print(f"[2/3] Dịch {len(segments)} đoạn sang tiếng Việt...") if src_lang == "vi": print(" Video đã là tiếng Việt, bỏ qua bước dịch.") return [t for _, _, t in segments] translator = GoogleTranslator(source="auto", target="vi") results = [] for i, (_, _, text) in enumerate(segments, start=1): try: vi = translator.translate(text) or text except Exception as e: # lỗi mạng / bị giới hạn -> giữ nguyên bản gốc print(f" Lỗi dịch đoạn {i}: {e}") vi = text results.append(vi) print(f" {i}/{len(segments)}: {vi}") return results def write_srt(segments, translated, srt_path: Path, bilingual: bool): with open(srt_path, "w", encoding="utf-8") as f: for i, ((start, end, original), vi) in enumerate(zip(segments, translated), start=1): f.write(f"{i}\n{format_time(start)} --> {format_time(end)}\n") f.write(f"{vi}\n") if bilingual and vi != original: f.write(f"{original}\n") f.write("\n") print(f" Đã lưu phụ đề: {srt_path}") def hex_to_ass(color: str) -> str: """'#ffeb3b' -> '&H003BEBFF' (ASS dùng thứ tự BGR).""" c = color.lstrip("#") r, g, b = c[0:2], c[2:4], c[4:6] return f"&H00{b}{g}{r}".upper().replace("&H", "&H") def merge_into_video(video: Path, srt: Path, out: Path, soft: bool, color: str = "#ffeb3b", size: int = 24, bold: bool = True): print("[3/3] Ghép phụ đề vào video...") if soft: # Phụ đề mềm: bật/tắt được trong trình phát, không mã hóa lại video -> rất nhanh cmd = [ "ffmpeg", "-y", "-i", str(video), "-i", str(srt), "-c", "copy", "-c:s", "mov_text", "-metadata:s:s:0", "language=vie", str(out), ] else: # Phụ đề cứng: ghi thẳng vào hình, mọi trình phát đều thấy srt_escaped = str(srt).replace("\\", "/").replace(":", "\\:") style = ( f"FontName=Arial,FontSize={size},Bold={1 if bold else 0}," f"PrimaryColour={hex_to_ass(color)},OutlineColour=&H00000000," "BorderStyle=1,Outline=2,Shadow=1,MarginV=30" ) cmd = [ "ffmpeg", "-y", "-i", str(video), "-vf", f"subtitles='{srt_escaped}':force_style='{style}'", "-c:a", "copy", str(out), ] subprocess.run(cmd, check=True) print(f" Xong: {out}") def main(): p = argparse.ArgumentParser(description="Tạo video phụ đề tiếng Việt từ video bất kỳ ngôn ngữ") p.add_argument("video", type=Path) p.add_argument("--model", default="small", help="tiny | base | small | medium | large-v3") p.add_argument("--device", default="cpu", choices=["cpu", "cuda"]) p.add_argument("--src", default=None, help="Mã ngôn ngữ gốc (en, ja, ko, zh...). Bỏ trống = tự phát hiện") p.add_argument("--bilingual", action="store_true", help="Hiện cả tiếng Việt và bản gốc") p.add_argument("--soft", action="store_true", help="Phụ đề mềm (nhanh, bật/tắt được) thay vì ghi cứng") p.add_argument("--color", default="#ffeb3b", help="Màu chữ phụ đề dạng hex, ví dụ #ffeb3b (vàng), #ffffff (trắng)") p.add_argument("--size", type=int, default=24, help="Cỡ chữ phụ đề khi ghi cứng (mặc định 24, thử 28-36 cho chữ to)") p.add_argument("--no-bold", action="store_true", help="Không in đậm") args = p.parse_args() srt_path = args.video.with_name(args.video.stem + "_vi.srt") out_path = args.video.with_name(args.video.stem + "_vi.mp4") segments, lang = transcribe(args.video, args.model, args.device, args.src) translated = translate_segments(segments, lang) write_srt(segments, translated, srt_path, args.bilingual) merge_into_video(args.video, srt_path, out_path, args.soft, args.color, args.size, not args.no_bold) if __name__ == "__main__": main()
